// ===== TẠO NỘI DUNG TĨNH CHO TỪNG LOẠI TRANG =====
function generateStaticBody(path: string, meta: typeof PAGE_META[string]): string {
  if (path === "/") return generateHomepageBody();
  if (path.startsWith("/du-an/")) return generateProjectBody(path);
  if (path.startsWith("/bat-dong-san-")) return generateGeoBody(path, meta);
  if (path === "/marketplace") return generateMarketplaceBody();
  return generateGenericBody(meta);
}

function escapeHtml(str: string): string {
  return str
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}
